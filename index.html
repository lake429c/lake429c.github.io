import random from math import sin, pi # Variables width, height = 0, 0 running = True fps = 10 x = 0 position = (0, 0) # Initialize color def initial_color(): return ( 255 * (1 + sin(0)) / 2, 255 * (1 + sin(2 * pi / 3)) / 2, 255 * (1 + sin(4 * pi / 3)) / 2 ) color_now = initial_color() # Update color function def update_color(x): r = 255 * (1 + sin(x)) / 2 g = 255 * (1 + sin(x + 2 * pi / 3)) / 2 b = 255 * (1 + sin(x + 4 * pi / 3)) / 2 return (r, g, b) # Initialize canvas def setup_canvas(): global width, height, position canvas = js.document.getElementById("canvas") width = js.window.innerWidth height = js.window.innerHeight canvas.width = width canvas.height = height position = (width // 2, height // 2) # Render function def render(): global x, color_now, position ctx = js.document.getElementById("canvas").getContext("2d") r, g, b = color_now # Draw pixel at current position ctx.fillStyle = f"rgb({int(r)}, {int(g)}, {int(b)})" ctx.fillRect(position[0], position[1], 1, 1) # Update position (randomly choose up, down, left, or right) direction = random.choice(['up', 'down', 'left', 'right']) if direction == 'up': position = (position[0], max(0, position[1] - 1)) elif direction == 'down': position = (position[0], min(height - 1, position[1] + 1)) elif direction == 'left': position = (max(0, position[0] - 1), position[1]) elif direction == 'right': position = (min(width - 1, position[0] + 1), position[1]) # Update color x += 1 color_now = update_color(x) # Toggle running state def toggle_running(event): nonlocal running running = not running # Main loop def loop(): if running: render() js.setTimeout(loop, 1000 // fps) # Event listeners canvas = js.document.getElementById("canvas") canvas.addEventListener("click", toggle_running) # Setup and start the loop setup_canvas() loop()
